#### 2.1.1 Zobrist哈希

**功能描述**: 为棋盘状态生成唯一哈希值，用于转置表和网络同步

**参考实现**:
- Egaroucid: `hash.hpp` - 使用 `uint32_t hash_rand_player[4][65536]`
- edax-reversi: `hash.h` - 完整的HashTable设计

**实现方案**:

```cpp
// src/ai/ZobristHash.h (参考 Egaroucid hash.hpp)

#pragma once

#include <cstdint>
#include <array>
#include "core/BitBoard.h"

namespace Reversi {

/**
 * @brief Zobrist哈希生成器
 *
 * 使用Egaroucid风格的Zobrist哈希方案：
 * - 4个player hash数组 (按位置和棋子数)
 * - 支持64x4位置编码
 *
 * 参考: Egaroucid/src/engine/hash.hpp
 */
class ZobristHash {
public:
    /**
     * @brief 初始化哈希表
     * @param hash_level 哈希精度级别 (25 = 2^25 entries)
     */
    static void init(int hash_level = 25);

    /**
     * @brief 生成棋盘哈希值
     * @param board 棋盘状态
     * @return 64位哈希值
     */
    static uint64_t computeHash(const BitBoard& board);

    /**
     * @brief 计算玩家与对手的哈希值
     * @param board 棋盘状态
     * @return 玩家哈希值
     */
    static uint64_t computePlayerHash(const BitBoard& board);

    /**
     * @brief 获取单格哈希值
     * @param square 格子索引 (0-63)
     * @param player 玩家 (0=黑, 1=白)
     * @return 哈希值
     */
    static uint64_t getSquareHash(int square, int player);

    /**
     * @brief 预计算的奇偶性表 (参考 edax-reversi parity计算)
     */
    static const uint64_t* getParityTable();

private:
    // Egaroucid风格: 4个数组，每个65536个条目
    static std::array<std::array<uint32_t, 65536>, 4> hash_player_;
    static std::array<std::array<uint32_t, 65536>, 4> hash_opponent_;
    static bool initialized_;
};

} // namespace Reversi
```

#### 2.1.2 转置表 (Transposition Table)

**功能描述**: 缓存搜索结果，避免重复计算

**参考实现**:
- edax-reversi: `hash.h` - `HashDraft`, `HashData`, `HashStore` 结构
- Egaroucid: `hash.hpp` - 完整的哈希表管理

**实现方案**:

```cpp
// src/ai/TranspositionTable.h

#pragma once

#include <cstdint>
#include <vector>
#include "core/BitBoard.h"
#include "ZobristHash.h"

namespace Reversi {

/**
 * @brief 转置表条目 (参考 edax-reversi hash.h)
 *
 * 存储搜索结果以避免重复计算
 */
struct TTEntry {
    uint64_t hash_;           // 棋盘哈希值
    int8_t depth_;            // 搜索深度
    int8_t score_;            // 评估分数
    int8_t alpha_;            // Alpha边界
    int8_t beta_;             // Beta边界
    Move move_;               // 最佳着法

    // 标志位
    bool valid_ : 1;
    bool exact_ : 1;          // 精确值 (不是alpha/beta边界)
    bool from_root_ : 1;     // 来自根节点

    /**
     * @brief 检查哈希是否匹配
     */
    bool match(uint64_t hash, int depth) const {
        return valid_ && hash_ == hash && depth_ >= depth;
    }
};

/**
 * @brief 转置表管理器
 *
 * 特性:
 * - 简单覆盖策略 (后续可升级为替换策略)
 * - 支持Alpha/Beta/Exact三种类型
 * - 可配置大小
 */
class TranspositionTable {
public:
    /**
     * @brief 构造函数
     * @param size_mb 转置表大小 (MB)
     */
    explicit TranspositionTable(size_t size_mb = 64);

    /**
     * @brief 清空转置表
     */
    void clear();

    /**
     * @brief 存储搜索结果
     * @param hash 棋盘哈希
     * @param depth 搜索深度
     * @param score 评估分数
     * @param alpha Alpha值
     * @param beta Beta值
     * @param move 最佳着法
     */
    void store(uint64_t hash, int depth, int score, int alpha, int beta, Move move);

    /**
     * @brief 查找条目
     * @param hash 棋盘哈希
     * @param depth 期望深度
     * @return 找到返回TTEntry，否则返回nullptr
     */
    const TTEntry* probe(uint64_t hash, int depth) const;

    /**
     * @brief 获取命中次数
     */
    uint64_t getHits() const { return hits_; }

    /**
     * @brief 获取查找次数
     */
    uint64_t getLookups() const { return lookups_; }

    /**
     * @brief 获取命中率
     */
    double getHitRate() const {
        return lookups_ > 0 ? static_cast<double>(hits_) / lookups_ : 0.0;
    }

private:
    std::vector<TTEntry> table_;
    size_t mask_;  // 索引掩码
    uint64_t hits_ = 0;
    uint64_t lookups_ = 0;
};

} // namespace Reversi
```

#### 2.1.4 Head-to-Head对战引擎

**功能描述**: 批量AI对战测试，生成胜率统计

**参考来源**: Egaroucid `bin/egaroucid_vs_edax.py`, `bin/battle.py`

**实现方案**:

```cpp
// src/research/BattleEngine.h

#pragma once

#include <memory>
#include <string>
#include "ai/AIStrategy.h"
#include "core/Board.h"
#include "Statistics.h"

namespace Reversi {

/**
 * @brief 对战配置
 */
struct BattleConfig {
    std::string player1_name;
    std::string player2_name;
    std::unique_ptr<AIStrategy> player1;
    std::unique_ptr<AIStrategy> player2;

    // 搜索限制
    SearchLimits limits1;
    SearchLimits limits2;

    // 对战设置
    int num_games = 50;                    // 对战局数
    bool alternate_first = true;            // 交替先手
    bool parallel = false;                  // 并行对战
};

/**
 * @brief 单场对战结果
 */
struct GameResult {
    int game_number;
    PlayerColor winner;       // 胜者
    int black_score;           // 黑棋最终棋子数
    int white_score;           // 白棋最终棋子数
    int moves_count;           // 回合数
    double duration_ms;        // 对战耗时 (毫秒)
};

/**
 * @brief 对战统计结果
 */
struct BattleStats {
    std::string player1_name;
    std::string player2_name;
    int total_games;
    int player1_wins;         // player1胜场
    int player2_wins;         // player2胜场
    int draws;                // 平局
    double win_rate1;         // player1胜率
    double win_rate2;         // player2胜率
    double avg_moves;         // 平均回合数
    double avg_duration_ms;   // 平均耗时

    // 按先手/后手统计
    int player1_first_wins;
    int player1_first_games;
    int player2_first_wins;
    int player2_first_games;

    // 统计显著性
    double p_value;           // Wilcoxon p值
    bool significant;         // 是否显著 (p < 0.05)
};

/**
 * @brief Head-to-Head对战引擎
 *
 * 参考: Egaroucid bin/battle.py, bin/egaroucid_vs_edax.py
 */
class BattleEngine {
public:
    /**
     * @brief 运行批量对战
     */
    static BattleStats runBattle(const BattleConfig& config);

    /**
     * @brief 运行单场对战
     */
    static GameResult playSingleGame(
        AIStrategy& player1,
        AIStrategy& player2,
        PlayerColor first_player,
        const SearchLimits& limits1,
        const SearchLimits& limits2
    );

    /**
     * @brief 导出对战结果到CSV
     */
    static void exportToCSV(const BattleStats& stats, const std::string& filepath);

    /**
     * @brief 导出详细对战记录
     */
    static void exportGameLog(
        const std::vector<GameResult>& games,
        const std::string& filepath
    );
};

} // namespace Reversi
```

#### 2.1.5 统计分析工具

**功能描述**: 统计显著性验证、置信区间计算

**参考来源**: Egaroucid `benchmark/` 目录下的分析脚本

**实现方案**:

```cpp
// src/research/Statistics.h

#pragma once

#include <vector>
#include <cmath>
#include <algorithm>
#include <numeric>

namespace Reversi {

/**
 * @brief 统计工具类
 *
 * 实现:
 * - 均值和标准差
 * - 95%置信区间
 * - Wilcoxon signed-rank测试
 * - 胜率比较
 */
class Statistics {
public:
    /**
     * @brief 计算均值
     */
    template<typename T>
    static double mean(const std::vector<T>& data);

    /**
     * @brief 计算标准差
     */
    template<typename T>
    static double stddev(const std::vector<T>& data);

    /**
     * @brief 计算95%置信区间
     */
    template<typename T>
    static std::pair<double, double> confidenceInterval95(const std::vector<T>& data);

    /**
     * @brief Wilcoxon signed-rank测试
     *
     * 用于配对样本的统计显著性检验
     *
     * @param sample1 样本1 (胜/负序列: +1/-1)
     * @param sample2 样本2
     * @return pair(统计量, p值)
     */
    static std::pair<double, double> wilcoxonSignedRankTest(
        const std::vector<int>& sample1,
        const std::vector<int>& sample2
    );

    /**
     * @brief 计算胜率及其置信区间
     */
    static double winRateWithCI(int wins, int total, double& ci_low, double& ci_high);

    /**
     * @brief 二项分布检验
     */
    static double binomialTest(int wins, int total, double expected_rate = 0.5);
};

// 实现
template<typename T>
double Statistics::mean(const std::vector<T>& data) {
    if (data.empty()) return 0.0;
    return static_cast<double>(
        std::accumulate(data.begin(), data.end(), 0LL)
    ) / data.size();
}

} // namespace Reversi
```
